films
----------
film_id BIGINT PK
name VARCHAR(255)
description VARCHAR(200) NULL
release_date DATE
duration BIGINT
rating_id INTEGER FK >- age_ratings.rating_id


genres
----------
genre_id BIGINT PK
name VARCHAR(255)

film_genres  # Связующая таблица для отношения многие ко многим
----------
film_id BIGINT FK >- films.film_id PK
genre_id BIGINT FK >- genres.genre_id PK

age_ratings
----------
rating_id INTEGER PK
rating VARCHAR(10)

film_likes
----------
film_id BIGINT FK >- films.film_id PK
user_id BIGINT FK >- users.user_id PK


users
----------
user_id BIGINT PK  # id
email VARCHAR(255)  # email
login VARCHAR(255)  # login 
name VARCHAR(255) NULL  # name (может быть NULL)
birthday DATE NULL  # birthday (может быть NULL)

friendships
----------
user_id BIGINT FK >- users.user_id PK # Идентификатор пользователя
person_id BIGINT FK >- users.person_id PK # Идентификатор другого пользователя
